==== PROGRAM 7 : MongoDB - comparison ($gt,$lte,$in,$ne) & logical ($and,$or) selectors ====

# 0. Make sure MongoDB server is running
#    Windows : net start MongoDB      (or start "MongoDB" service)
#    Linux   : sudo systemctl start mongod
#    Mac     : brew services start mongodb-community

# 1. Option A - run the whole script at once
cd fsadlab/p7
mongosh fsadlab_db --file employees.js

# 2. Option B - interactive
mongosh
  use fsadlab_db
  show dbs
  show collections
  # paste the insertMany(...) from employees.js, then run queries e.g.:
  db.employees.find({ salary: { $gt: 90000, $lte: 120000 } })
  db.employees.find({ department: { $in: ["HR","Sales"] } })
  db.employees.find({ status: { $ne: "On Leave" } })
  db.employees.find({ department: "Engineering", performance_score: { $gte: 4.5 } })
  db.employees.find({ $or: [ {department:"HR"}, {salary:{$gt:100000}} ] })
  db.employees.find({ status:"Active", department:{$in:["Engineering","Sales"]}, hire_date:{$lt:new Date("2024-01-01")} })
  db.employees.find().pretty()
  db.employees.countDocuments()
  db.employees.drop()          # clean up
  exit
